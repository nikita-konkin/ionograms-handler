# What the training container installs: everything inference has, plus the
# things only fitting needs.
#
# Separate from requirements-infer.txt rather than a superset flag, because the
# split is the point: `infer` runs unattended on a timer and should pull as
# little as possible, while `train` is started by hand and can afford to be
# large.

-r requirements-infer.txt

# Hyperparameter search.
hyperopt>=0.2.7

# SARIMA, and the auto-order search around it.
pmdarima>=2.0

# Deep models. `tensorflow-cpu` rather than `tensorflow`: there is no GPU on
# the work server, and the CUDA wheels are several times the size for
# libraries that would never load.
#
# `tf-keras` alongside it because the archive holds BOTH Keras formats --
# `nn_tuner.py` writes `save_format="tf"` (a Keras 2 SavedModel, which Keras 3
# cannot read) while newer files are the Keras 3 `.keras` zip. With
# TF_USE_LEGACY_KERAS=1 set, the old ones load; without it, the new ones do.
# See muf/extractors/cnn.py, which documents the same trap for the autoencoder.
tensorflow-cpu>=2.16,<3.0
tf-keras>=2.16
